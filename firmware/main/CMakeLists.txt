  if(NOT AIRBRIDGE_IPV6_LINK OR NOT AIRBRIDGE_PRESENCE_TX OR (AIRBRIDGE_PROBE_PEER STREQUAL "" AND NOT AIRBRIDGE_PUBLIC_BUILD))

  endif()
  target_sources(${COMPONENT_LIB} PRIVATE "native_receiver.cpp" "native_sender.cpp" "device_identity.cpp" "ble_presence.cpp" "serial_channel.cpp" "../../core/archive.cpp" "../../core/native_service.cpp" "../../core/serial_protocol.cpp")

  target_compile_definitions(${COMPONENT_LIB} PRIVATE AIRBRIDGE_NATIVE_SERVICE=1)

if(AIRBRIDGE_SCAN_CAPTURE)
  target_compile_definitions(${COMPONENT_LIB} PRIVATE AIRBRIDGE_SCAN_CAPTURE=1)
endif()
set(AIRBRIDGE_PROBE_PEER "" CACHE STRING "Explicit MAC for one bounded Discover qualification")
if(NOT AIRBRIDGE_PROBE_PEER STREQUAL "" OR AIRBRIDGE_PUBLIC_BUILD)
  if(NOT AIRBRIDGE_IPV6_LINK OR NOT AIRBRIDGE_PRESENCE_TX)
    message(FATAL_ERROR "Discover qualification requires presence and IPv6 link")
  endif()
  if(AIRBRIDGE_PUBLIC_BUILD)
    if(NOT AIRBRIDGE_PROBE_PEER STREQUAL "")
      message(FATAL_ERROR "Public firmware cannot contain a private peer address")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE AIRBRIDGE_PUBLIC_BUILD=1)
    set(AIRBRIDGE_PROBE_PEER "00:00:00:00:00:00")
  endif()
  string(REPLACE ":" ";" probe_octets "${AIRBRIDGE_PROBE_PEER}")
  list(LENGTH probe_octets probe_count)
  if(NOT probe_count EQUAL 6)
    message(FATAL_ERROR "Probe target must be a six-octet MAC")
  endif()
  set(probe_bytes "")
  foreach(octet IN LISTS probe_octets)
    if(NOT octet MATCHES "^[0-9a-fA-F][0-9a-fA-F]$")
      message(FATAL_ERROR "Invalid probe MAC octet")
    endif()
    list(APPEND probe_bytes "0x${octet}")
  endforeach()
  string(JOIN "," probe_initializer ${probe_bytes})
  target_sources(${COMPONENT_LIB} PRIVATE "discovery_probe.cpp" "../../core/bplist.cpp" "../../core/http.cpp")
  target_compile_definitions(${COMPONENT_LIB} PRIVATE AIRBRIDGE_DISCOVER_PROBE=1 "AIRBRIDGE_PROBE_PEER_BYTES=${probe_initializer}")
endif()
