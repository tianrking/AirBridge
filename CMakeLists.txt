cmake_minimum_required(VERSION 3.20)
project(airbridge_protocol LANGUAGES CXX)
add_library(airbridge_protocol core/awdl.cpp)
target_include_directories(airbridge_protocol PUBLIC core)
target_compile_features(airbridge_protocol PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(airbridge_protocol PRIVATE /W4 /WX)
else()
  target_compile_options(airbridge_protocol PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE airbridge_protocol)
add_test(NAME protocol_tests COMMAND protocol_tests)
